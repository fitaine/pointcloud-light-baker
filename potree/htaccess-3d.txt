# COPC .laz files require Range requests — Apache supports these natively.
# CORS headers are needed so the COPC JS library (fetch from the gallery page) can read the file.

Header set Access-Control-Allow-Origin "*"
Header set Access-Control-Allow-Headers "Range"
Header set Access-Control-Expose-Headers "Content-Range, Accept-Ranges"
Header set Accept-Ranges "bytes"

# Cache point clouds and volumes forever — they never change once deployed.
<FilesMatch "\.(laz|bin)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
</FilesMatch>

# Don't cache the viewer HTML and JS — allow updates.
<FilesMatch "\.(html|js)$">
    Header set Cache-Control "no-store, no-cache, must-revalidate"
</FilesMatch>

# Prevent directory listing.
Options -Indexes
